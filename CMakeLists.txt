cmake_minimum_required(VERSION 3.20)
if(CMAKE_HOST_APPLE AND NOT DEFINED CMAKE_OSX_DEPLOYMENT_TARGET)
    set(CMAKE_OSX_DEPLOYMENT_TARGET "13.0" CACHE STRING "Earliest supported macOS version")
endif()
project(MaxDisplay VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()

if(POLICY CMP0135)
    cmake_policy(SET CMP0135 NEW)
endif()
include(cmake/Dependencies.cmake)
if(MSVC)
    add_compile_options("$<$<COMPILE_LANGUAGE:CXX>:/utf-8>")
endif()
find_package(Threads REQUIRED)
add_library(transport STATIC src/net.cpp src/wire.cpp src/udp.cpp)
target_include_directories(transport PUBLIC src)
target_link_libraries(transport PUBLIC Threads::Threads PRIVATE enet)
if(WIN32)
    target_compile_definitions(transport PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE _WIN32_WINNT=0x0A00 WINVER=0x0A00)
    target_link_libraries(transport PUBLIC ws2_32)
endif()

if(APPLE)
    enable_language(OBJCXX)
    set(CMAKE_OBJCXX_STANDARD 17)
    set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
    add_compile_options("$<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>")
    add_library(video_codec STATIC src/mac/encoder.mm src/av1.cpp)
    target_link_libraries(video_codec PUBLIC "-framework CoreFoundation" "-framework Foundation" "-framework VideoToolbox" "-framework CoreMedia" "-framework CoreVideo")
    add_library(virtual_display STATIC src/mac/virtual_display.mm)
    target_include_directories(virtual_display PUBLIC src)
    target_link_libraries(virtual_display PUBLIC "-framework Foundation" "-framework CoreGraphics" "-framework CoreFoundation")
    add_library(mac_sender STATIC src/mac/sender.mm)
    target_link_libraries(mac_sender PUBLIC transport video_codec virtual_display "-framework ScreenCaptureKit")
    add_executable(maxdisplay src/main.cpp src/cli.cpp)
    target_link_libraries(maxdisplay PRIVATE mac_sender)
elseif(WIN32)
    add_library(video_codec STATIC src/win/decoder.cpp src/win/hevc_software.cpp src/av1.cpp)
    target_link_libraries(video_codec PRIVATE de265)
    target_include_directories(video_codec PRIVATE "${libde265_SOURCE_DIR}" "${libde265_BINARY_DIR}")
    target_compile_definitions(video_codec PRIVATE LIBDE265_STATIC_BUILD)
    target_compile_definitions(video_codec PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE _WIN32_WINNT=0x0A00 WINVER=0x0A00)
    target_link_libraries(video_codec PUBLIC mfplat mfuuid wmcodecdspuuid ole32 d3d11 dxgi dxguid)
    add_library(win_receiver STATIC src/win/receiver.cpp src/win/display_info.cpp)
    target_link_libraries(win_receiver PUBLIC transport video_codec user32 gdi32 iphlpapi)
    add_library(windows_entry OBJECT src/main.cpp src/cli.cpp)
    target_link_libraries(windows_entry PRIVATE win_receiver)
    target_include_directories(windows_entry PRIVATE "${PROJECT_BINARY_DIR}/generated")
    add_executable(maxdisplay $<TARGET_OBJECTS:windows_entry>)
    target_link_libraries(maxdisplay PRIVATE win_receiver)
    if(MSVC)
        add_custom_target(windows_relink_kit
            COMMAND "${CMAKE_COMMAND}" "-DBUILD_ROOT=${PROJECT_BINARY_DIR}" "-DHEVC_SOURCE=${libde265_SOURCE_DIR}"
                    "-DGENERATOR=${CMAKE_GENERATOR}" "-DPLATFORM=${CMAKE_GENERATOR_PLATFORM}" "-DCONFIG=$<CONFIG>"
                    "-DAPP_OBJECTS=$<JOIN:$<TARGET_OBJECTS:windows_entry>,|>"
                    "-DAPP_LIBRARIES=$<TARGET_FILE:win_receiver>|$<TARGET_FILE:video_codec>|$<TARGET_FILE:transport>|$<TARGET_FILE:aom>|$<TARGET_FILE:enet>"
                    -P "${PROJECT_SOURCE_DIR}/cmake/WindowsRelink.cmake"
            DEPENDS maxdisplay VERBATIM)
    endif()
else()
    message(STATUS "The display app targets macOS and Windows. Building the portable transport only.")
endif()

if(TARGET video_codec)
    target_include_directories(video_codec PUBLIC src PRIVATE "${libaom_SOURCE_DIR}")
    target_link_libraries(video_codec PRIVATE aom transport)
    target_include_directories(maxdisplay PRIVATE "${PROJECT_BINARY_DIR}/generated")
endif()

include(CTest)
if(BUILD_TESTING)
    foreach(test_name IN ITEMS rate_control recovery)
        add_executable(${test_name}_tests tests/${test_name}.cpp)
        target_include_directories(${test_name}_tests PRIVATE src)
        add_test(NAME ${test_name} COMMAND ${test_name}_tests)
    endforeach()
    add_executable(display_geometry_tests tests/display_geometry.cpp)
    target_link_libraries(display_geometry_tests PRIVATE transport)
    add_test(NAME display_geometry COMMAND display_geometry_tests)
    add_executable(transport_tests tests/transport.cpp)
    target_link_libraries(transport_tests PRIVATE transport)
    add_test(NAME transport COMMAND transport_tests)
    set_tests_properties(transport PROPERTIES TIMEOUT 30)
    add_executable(udp_tests tests/udp.cpp)
    target_link_libraries(udp_tests PRIVATE transport)
    add_test(NAME udp_transport COMMAND udp_tests)
    set_tests_properties(udp_transport PROPERTIES TIMEOUT 35)
    if(TARGET maxdisplay)
        add_executable(av1_rungs_tests tests/av1_rungs.cpp)
        target_link_libraries(av1_rungs_tests PRIVATE video_codec transport)
        add_test(NAME av1_duration_matches_rung COMMAND av1_rungs_tests)
        add_executable(cli_options_tests src/cli.cpp tests/cli_options.cpp)
        target_include_directories(cli_options_tests PRIVATE "${PROJECT_BINARY_DIR}/generated")
        target_link_libraries(cli_options_tests PRIVATE transport)
        add_test(NAME cli_options COMMAND cli_options_tests)
        add_test(NAME cli_help COMMAND maxdisplay --help)
        add_executable(codec_probe tests/codec_probe.cpp)
        target_link_libraries(codec_probe PRIVATE video_codec transport)
    endif()
endif()

if(APPLE AND BUILD_TESTING)
    # Substitute only the hardware submission call in this test-only encoder object.
    # Its replacement forwards software submissions to the real VideoToolbox framework.
    add_library(mac_fallback_encoder OBJECT src/mac/encoder.mm)
    target_include_directories(mac_fallback_encoder PRIVATE src)
    target_compile_definitions(mac_fallback_encoder PRIVATE VTCompressionSessionEncodeFrame=TestVTEncodeFrame)
    add_executable(mac_fallback_tests tests/mac_fallback.cpp $<TARGET_OBJECTS:mac_fallback_encoder>)
    target_link_libraries(mac_fallback_tests PRIVATE video_codec)
    foreach(codec IN ITEMS h264 h265)
        add_test(NAME mac_hardware_fallback_${codec} COMMAND mac_fallback_tests ${codec})
        set_tests_properties(mac_hardware_fallback_${codec} PROPERTIES TIMEOUT 30 SKIP_RETURN_CODE 77)
    endforeach()

    add_executable(virtual_display_tests tests/virtual_display.mm)
    target_link_libraries(virtual_display_tests PRIVATE virtual_display)
    add_test(NAME virtual_display_lifecycle COMMAND virtual_display_tests)
    set_tests_properties(virtual_display_lifecycle PROPERTIES TIMEOUT 30 SKIP_RETURN_CODE 77)

    add_executable(mac_capture_tests tests/mac_capture.cpp)
    target_link_libraries(mac_capture_tests PRIVATE mac_sender)
    add_test(NAME mac_live_capture_tcp COMMAND mac_capture_tests h264-tcp)
    add_test(NAME mac_live_capture_udp COMMAND mac_capture_tests av1-udp)
    add_test(NAME mac_live_capture_hevc COMMAND mac_capture_tests h265-udp)
    set_tests_properties(mac_live_capture_tcp mac_live_capture_udp mac_live_capture_hevc PROPERTIES TIMEOUT 90 SKIP_RETURN_CODE 77)
    set_tests_properties(virtual_display_lifecycle mac_live_capture_tcp mac_live_capture_udp mac_live_capture_hevc
                         PROPERTIES RESOURCE_LOCK quartz_display)
endif()

if(WIN32 AND BUILD_TESTING)
    add_executable(windows_info_tests tests/windows_info.cpp)
    target_link_libraries(windows_info_tests PRIVATE win_receiver)
    add_test(NAME windows_info COMMAND windows_info_tests)
    add_executable(windows_present_tests tests/windows_present.cpp)
    target_link_libraries(windows_present_tests PRIVATE win_receiver)
endif()
